Base URL & Authentication

Base API Endpoint
https://api.smsroutebd.com/v1
Authorization Header
Authorization: Bearer YOUR_API_KEY
POST /v1/sms/send
Dispatch Single SMS instantly

Sends a high-priority message to any Bangladeshi phone number (supports numbers with or without +88 country code). 

curl -X POST https://api.smsroutebd.com/v1/sms/send \
  -H "Authorization: Bearer sms_live_9a7f34c2b8109d4e5f6a7b8c9d0e1f2a" \
  -H "Content-Type: application/json" \
  -d '{
    "recipient": "01923988380",
    "message": "Your verification code is 492018",
    "sender_id": "SMS ROUTE BD"
  }'
<?php
$ch = curl_init('https://api.smsroutebd.com/v1/sms/send');
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Authorization: Bearer sms_live_9a7f34c2b8109d4e5f6a7b8c9d0e1f2a',
    'Content-Type: application/json'
]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
    'recipient' => '01923988380',
    'message' => 'Your verification code is 492018',
    'sender_id' => 'SMS ROUTE BD'
]));

$response = curl_exec($ch);
curl_close($ch);
echo $response;
?>
